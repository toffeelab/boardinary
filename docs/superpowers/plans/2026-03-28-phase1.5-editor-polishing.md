- [ ] **Step 1: use-edge-validation.ts**

```ts
import { useCallback } from "react";
import type { Edge, Connection } from "@xyflow/react";

function hasCycle(edges: Edge[], source: string, target: string): boolean {
  const adj = new Map<string, string[]>();
  for (const edge of edges) {
    if (!adj.has(edge.source)) adj.set(edge.source, []);
    adj.get(edge.source)!.push(edge.target);
  }
  // 새 엣지 추가 후 target에서 source로 도달 가능하면 사이클
  const visited = new Set<string>();
  const stack = [target];
  while (stack.length > 0) {
    const node = stack.pop()!;
    if (node === source) return true;
    if (visited.has(node)) continue;
    visited.add(node);
    const neighbors = adj.get(node) ?? [];
    stack.push(...neighbors);
  }
  return false;
}

export function useEdgeValidation(edges: Edge[]) {
  const isValidConnection = useCallback(
    (connection: Connection) => {
      // 자기 자신 연결 방지
      if (connection.source === connection.target) return false;

      // 중복 연결 방지
      const isDuplicate = edges.some(
        (e) =>
          e.source === connection.source &&
          e.target === connection.target &&
          e.sourceHandle === connection.sourceHandle,
      );
      if (isDuplicate) return false;

      // 순환 참조 방지
      if (hasCycle(edges, connection.source!, connection.target!)) return false;

      return true;
    },
    [edges],
  );

  return { isValidConnection };
}
```
